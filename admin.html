Lihat & Salin Skema SQL Lengkap Arsitektur Supabase Ravenmille

Salin skrip SQL di bawah dan jalankan pada SQL Editor di dashboard Supabase Anda untuk membuat seluruh tabel, kebijakan akses (RLS), serta kolom foto profil yang terhubung dengan kode aplikasi ini: 

-- =========================================================
-- Ravenmille – Skema Database & Storage Supabase Lengkap
-- =========================================================
create extension if not exists "pgcrypto";

-- 1. Tabel Katalog Kendaraan
create table if not exists public.vehicles (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  brand text default '',
  category text default 'Katalog',
  price numeric not null default 0,
  fuel_eff numeric not null default 12,
  fuel_price numeric default 12950,
  km_year integer default 15000,
  service numeric default 0,
  insurance numeric default 0,
  tax numeric default 0,
  image_url text,
  sort_order integer default 0,
  is_active boolean default true,
  created_at timestamptz default now()
);

-- 2. Tabel Masalah Khas Kendaraan
create table if not exists public.vehicle_issues (
  id uuid primary key default gen_random_uuid(),
  vehicle_id uuid references public.vehicles(id) on delete cascade,
  name text not null,
  km text default '',
  cost text default '',
  symptom text default '',
  solution text default '',
  image_url text,
  created_at timestamptz default now()
);

-- 3. Tabel Inspektor Independen (+ Foto Profil)
create table if not exists public.inspectors (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  title text default 'Inspektor Independen Terdaftar',
  city text default '',
  province text default '',
  phone text default '',
  specialties text[] default '{}',
  tools text default '',
  price1 text default 'Konsultasi via WA',
  price2 text default '',
  availability text default 'Sesuai Janji',
  rating numeric,
  inspections integer default 0,
  image_url text,
  created_at timestamptz default now()
);
alter table public.inspectors add column if not exists image_url text;

-- 4. Tabel Bengkel Spesialis (+ Foto Profil)
create table if not exists public.workshops (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  category text default 'Bengkel Spesialis',
  city text default '',
  address text default '',
  hours text default '',
  phone text default '',
  maps text default '',
  warranty text default '',
  services text[] default '{}',
  image_url text,
  created_at timestamptz default now()
);
alter table public.workshops add column if not exists image_url text;

-- 5. Tabel Berita & Artikel
create table if not exists public.news (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  category text default 'BERITA',
  excerpt text default '',
  content text not null,
  image_url text,
  is_active boolean default true,
  created_at timestamptz default now()
);

-- 6. Tabel Konsultasi Pengunjung (Beranda)
create table if not exists public.consultations (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  email text not null,
  vehicle_interest text not null,
  goal text default 'beli-bekas',
  created_at timestamptz default now()
);

-- 7. Tabel Log Simulasi TCO, Risiko, & Komparasi (Opsional / Analitik)
create table if not exists public.tco_simulations (
  id uuid primary key default gen_random_uuid(),
  report_code text not null,
  vehicle_name text not null,
  price numeric default 0,
  period_years integer default 3,
  payment_method text default 'kredit',
  km_per_year integer default 15000,
  total_tco numeric default 0,
  monthly_cost numeric default 0,
  created_at timestamptz default now()
);

create table if not exists public.risk_assessments (
  id uuid primary key default gen_random_uuid(),
  vehicle_name text not null,
  vehicle_year integer,
  odometer_km integer,
  transmission text,
  body_type text,
  service_history text,
  incident_history text,
  risk_score integer,
  risk_level text,
  created_at timestamptz default now()
);

create table if not exists public.comparison_logs (
  id uuid primary key default gen_random_uuid(),
  period_years integer default 3,
  vehicle_a_name text not null,
  vehicle_a_price numeric default 0,
  vehicle_b_name text not null,
  vehicle_b_price numeric default 0,
  created_at timestamptz default now()
);

-- 8. Aktifkan RLS & Kebijakan Akses Publik (Sesuaikan untuk produksi)
alter table public.vehicles enable row level security;
alter table public.vehicle_issues enable row level security;
alter table public.inspectors enable row level security;
alter table public.workshops enable row level security;
alter table public.news enable row level security;
alter table public.consultations enable row level security;
alter table public.tco_simulations enable row level security;
alter table public.risk_assessments enable row level security;
alter table public.comparison_logs enable row level security;

create policy "public_all_vehicles" on public.vehicles for all using (true) with check (true);
create policy "public_all_issues" on public.vehicle_issues for all using (true) with check (true);
create policy "public_all_inspectors" on public.inspectors for all using (true) with check (true);
create policy "public_all_workshops" on public.workshops for all using (true) with check (true);
create policy "public_all_news" on public.news for all using (true) with check (true);
create policy "public_insert_select_consultations" on public.consultations for all using (true) with check (true);
create policy "public_insert_tco_logs" on public.tco_simulations for insert with check (true);
create policy "public_insert_risk_logs" on public.risk_assessments for insert with check (true);
create policy "public_insert_comparison_logs" on public.comparison_logs for insert with check (true);

-- 9. Storage Bucket Gambar ('vehicle-images')
insert into storage.buckets (id, name, public)
values ('vehicle-images', 'vehicle-images', true)
on conflict (id) do nothing;

create policy "public_read_vehicle_images" on storage.objects
  for select using (bucket_id = 'vehicle-images');
create policy "public_upload_vehicle_images" on storage.objects
  for insert with check (bucket_id = 'vehicle-images');
